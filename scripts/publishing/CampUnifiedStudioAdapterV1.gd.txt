func _ready() -> void:
    super._ready()
    if building_picker == null or building_controls == null or preview == null:
        push_error("CAMP STUDIO: original unified EDIFICIOS/viewport missing")
        return
    if building_picker.item_count != FIRST_CAMP_INDEX:
        push_error("CAMP STUDIO: seven existing buildings changed; refusing to replace catalogue")
        return
    for i in range(CAMP_KEYS.size()):
        building_picker.add_item(CAMP_LABELS[i])
        building_picker.set_item_metadata(FIRST_CAMP_INDEX + i,
            "camp:" + CAMP_KEYS[i])
    _camp_create_controls()
    camp_controls.visible = false
    _destroyer_studio_create_controls()
    if OS.has_feature("web"):
        call_deferred("_destroyer_studio_select_from_url")
    # Deep links open a camp INSIDE the one original Pages 3D scene:
    # ?campamento=medico (also suministros, reconocimiento, defensivo).
    if OS.has_feature("web"):
        call_deferred("_camp_select_from_url")

func _camp_select_from_url() -> void:
    if not OS.has_feature("web"):
        return
    var requested: Variant = JavaScriptBridge.eval(
        "(new URLSearchParams(window.location.search)).get('campamento') || ''",
        true)
    var key: String = str(requested).to_lower().strip_edges()
    if not key in CAMP_KEYS:
        return
    if category_picker == null or building_picker == null:
        return
    var item_index: int = FIRST_CAMP_INDEX + CAMP_KEYS.find(key)
    category_picker.select(1)
    _switch_category(1)
    building_picker.select(item_index)
    _change_selected_building(item_index)

func _camp_selected(index: int) -> bool:
    return index >= FIRST_CAMP_INDEX and (
        index < FIRST_CAMP_INDEX + CAMP_KEYS.size())

func _camp_hide_prior_controls() -> void:
    if not _camp_saved_visibility.is_empty():
        return
    for item in building_controls.get_children():
        if item is Control and item != camp_controls:
            var control: Control = item as Control
            if control == building_picker or control.is_ancestor_of(building_picker):
                continue
            _camp_saved_visibility[control] = control.visible
            control.visible = false
    camp_controls.visible = true
    if faction_picker != null:
        faction_picker.visible = false

func _camp_restore_prior_controls() -> void:
    _camp_stop_medical()
    _camp_stop_defense()
    _camp_stop_recon()
    _camp_stop_fx()
    _camp_stop_service()
    for item in _camp_saved_visibility.keys():
        if is_instance_valid(item):
            var control: Control = item as Control
            control.visible = bool(_camp_saved_visibility[item])
    _camp_saved_visibility.clear()
    if camp_controls != null:
        camp_controls.visible = false

func _change_selected_building(index: int) -> void:
    _destroyer_restore_generic_faction()
    _destroyer_studio_stop()
    _destroyer_studio_restore_floor()
    if is_instance_valid(_destroyer_water):
        _destroyer_water.hide()
    if is_instance_valid(_destroyer_inspection_light):
        _destroyer_inspection_light.hide()
    if is_instance_valid(_destroyer_lower_fill):
        _destroyer_lower_fill.hide()
    if destroyer_controls != null:
        destroyer_controls.hide()
    if _camp_selected(index) and camp_controls != null:
        camp_kind = CAMP_KEYS[index - FIRST_CAMP_INDEX]
        _camp_hide_prior_controls()
        _camp_display()
        return
    _camp_restore_prior_controls()
    super._change_selected_building(index)

func _switch_category(index: int) -> void:
    if index != 0:
        _destroyer_restore_generic_faction()
        _destroyer_studio_stop()
        _destroyer_studio_restore_floor()
    super._switch_category(index)
    if camp_controls == null or building_picker == null:
        return
    if index == 1 and _camp_selected(building_picker.selected):
        camp_kind = CAMP_KEYS[building_picker.selected - FIRST_CAMP_INDEX]
        _camp_hide_prior_controls()
        _camp_display()
    else:
        _camp_restore_prior_controls()
    _destroyer_studio_sync_visibility()

func _camp_option_changed(_index: int) -> void:
    if _role.begins_with("camp_"):
        _camp_display()

func _camp_display() -> void:
    if preview == null or building_picker == null or camp_controls == null:
        return
    if not _camp_selected(building_picker.selected):
        return
    # Keep the user's close-up inspection angle when swapping faction,
    # mobile LOD or damage on the SAME camp. Switching camp type resets it.
    var previous_kind: String = str(model_root.get_meta(
        "camp_kind", "")) if model_root != null else ""
    var same_camp: bool = previous_kind == camp_kind
    var previous_interior: bool = same_camp and bool(model_root.get_meta("suministros_v43_interior_open", false))
    var previous_scan: float = float(model_root.get_meta("recon_v2_visual_scan_degrees", 0.0)) if same_camp and camp_kind == "reconocimiento" else 0.0
    var previous_medical_open: bool = same_camp and bool(model_root.get_meta("medical_v3_inspection", false))
    var previous_yaw: float = _yaw
    var previous_zoom: float = _zoom
    var previous_deploy: float = float(model_root.get_meta(
        "visual_deploy_progress", 1.0)) if same_camp else 1.0
    var previous_defense_timeline: float = float(model_root.get_meta("defense_v33_timeline", -1.0)) if same_camp and camp_kind == "defensivo" else -1.0
    _camp_stop_medical()
    _camp_stop_defense()
    _camp_stop_recon()
    _camp_stop_fx()
    _camp_stop_service()
    _stop_animation()
    if model_root != null:
        if model_root.get_parent() != null:
            model_root.get_parent().remove_child(model_root)
        model_root.queue_free()
        model_root = null
    _role = "camp_" + camp_kind
    var faction: int = camp_faction_picker.selected - 1
    var lod: int = 8 if camp_lod_picker.selected == 0 else 13
    var damage: int = camp_damage_picker.selected
    camp_finish_picker.visible = camp_kind == "suministros"
    camp_recon_finish.visible = camp_kind == "reconocimiento"
    camp_defense_finish.visible = camp_kind == "defensivo"
    camp_medical_finish.visible = camp_kind == "medico"
    camp_medical_actions.visible = camp_kind == "medico" and camp_medical_finish.selected == 0 and damage < 2
    camp_defense_actions.visible = camp_kind == "defensivo" and camp_defense_finish.selected == 0 and damage < 2
    camp_recon_actions.visible = camp_kind == "reconocimiento" and damage < 2
    camp_recon_operation.visible = camp_recon_finish.selected == 0
    camp_damage_actions.visible = (camp_kind == "suministros" and camp_finish_picker.selected == 0) or (camp_kind == "reconocimiento" and camp_recon_finish.selected == 0) or (camp_kind == "defensivo" and camp_defense_finish.selected == 0) or (camp_kind == "medico" and camp_medical_finish.selected == 0)
    camp_supply_actions.visible = camp_kind == "suministros" and camp_finish_picker.selected == 0 and damage < 2
    model_root = CampArt.build(camp_kind, faction, lod, damage, camp_finish_picker.selected == 0, camp_recon_finish.selected == 0, camp_defense_finish.selected == 0, camp_medical_finish.selected == 0)
    if model_root == null:
        push_error("CAMP STUDIO: cannot assemble actual camp " + camp_kind)
        camp_status.text = "ERROR: modelo nativo no disponible"
        return
    model_root.name = "CampamentoUnificado_" + camp_kind
    model_root.set_meta("camp_shared_studio", true)
    preview.add_child(model_root)
    _yaw = previous_yaw if same_camp else -32.0
    _zoom = previous_zoom if same_camp else 1.0
    model_root.rotation_degrees.y = _yaw
    if same_camp and not CampArt.pose_deploy(model_root, previous_deploy):
        push_error("CAMP STUDIO: cannot restore current deployment state")
    if same_camp and camp_kind == "defensivo" and damage < 2 and camp_defense_finish.selected == 0 and previous_defense_timeline >= 0.0:
        DefenseMotion.deployment_frame(model_root, previous_defense_timeline)
    if previous_medical_open and damage < 2 and camp_kind == "medico" and camp_medical_finish.selected == 0:
        MedicalSculpt.inspect_interior(model_root, true)
    if previous_interior and damage < 2 and camp_kind == "suministros" and camp_finish_picker.selected == 0:
        SupplyV43.inspect_interior(model_root, true)
    if same_camp and damage < 2 and camp_kind == "reconocimiento":
        ReconOptics.pose_scan(model_root, previous_scan)
    camp_controls.visible = true
    _camp_hide_prior_controls()
    _update_camera()
    _update_details()
    _refresh_unified_studio_navigation()

func _camp_stop_recon() -> void:
    if _camp_recon_tween != null and _camp_recon_tween.is_valid():
        _camp_recon_tween.kill()
    _camp_recon_tween = null
    if is_instance_valid(model_root) and _role == "camp_reconocimiento":
        ReconStation.station_pose(model_root, 0.0)

func _camp_scan(angle: float) -> void:
    _camp_stop_fx()
    _camp_stop_recon()
    if model_root != null and _role == "camp_reconocimiento":
        ReconOptics.pose_scan(model_root, angle)

func _camp_recon_frame(progress: float) -> void:
    if is_instance_valid(model_root) and _role == "camp_reconocimiento":
        if ReconStation.station_pose(model_root, progress):
            ReconOptics.pose_scan(model_root, 85.0 * sin(progress * TAU))

func _camp_recon_operate() -> void:
    if not is_instance_valid(model_root) or _role != "camp_reconocimiento":
        return
    if not ReconStation.station_pose(model_root, 0.0):
        return
    _camp_stop_fx()
    _camp_stop_recon()
    _camp_recon_tween = create_tween()
    _camp_recon_tween.tween_method(_camp_recon_frame, 0.0, 1.0, 3.2)

func _camp_stop_defense() -> void:
    if _camp_defense_tween != null and _camp_defense_tween.is_valid():
        _camp_defense_tween.kill()
    _camp_defense_tween = null

func _camp_defense_frame(value: float) -> void:
    if is_instance_valid(model_root) and _role == "camp_defensivo":
        if DefenseMotion.deployment_frame(model_root, value):
            _update_details()

func _camp_defense_animate(target: float) -> void:
    if not is_instance_valid(model_root) or _role != "camp_defensivo" or model_root.get_meta("defense_art_v3", 0) != 3 or int(model_root.get_meta("visual_damage", 0)) >= 2:
        return
    _camp_stop_fx()
    _camp_stop_defense()
    var start: float = float(model_root.get_meta("defense_v33_timeline", 1.0))
    if absf(target - start) < 0.001:
        return
    _camp_defense_tween = create_tween()
    _camp_defense_tween.tween_method(_camp_defense_frame, start, target, maxf(0.12, absf(target - start) * 3.2))

func _camp_stop_medical() -> void:
    if _camp_medical_tween != null and _camp_medical_tween.is_valid():
        _camp_medical_tween.kill()
    _camp_medical_tween = null

func _camp_medical_frame(value: float) -> void:
    if is_instance_valid(model_root) and _role == "camp_medico":
        if CampArt.pose_deploy(model_root, value):
            _update_details()

func _camp_medical_animate(target: float) -> void:
    if not is_instance_valid(model_root) or _role != "camp_medico" or model_root.get_meta("medical_art_revision", "") != "3.5" or int(model_root.get_meta("visual_damage", 0)) >= 2:
        return
    _camp_stop_medical()
    MedicalSculpt.inspect_interior(model_root, false)
    var start: float = float(model_root.get_meta("visual_deploy_progress", 1.0))
    if absf(target - start) < 0.001:
        return
    _camp_medical_tween = create_tween()
    _camp_medical_tween.tween_method(_camp_medical_frame, start, target, maxf(0.12, absf(target - start) * 3.6))

func _camp_medical_monitor_frame(value: float) -> void:
    if is_instance_valid(model_root) and _role == "camp_medico":
        MedicalSculpt.monitor_frame(model_root, value)

func _camp_medical_monitor() -> void:
    if not is_instance_valid(model_root) or _role != "camp_medico" or not MedicalSculpt.monitor_frame(model_root, 0.0):
        return
    _camp_stop_medical()
    _camp_medical_tween = create_tween()
    _camp_medical_tween.tween_method(_camp_medical_monitor_frame, 0.0, 1.0, 2.4)

func _camp_medical_interior(enabled: bool) -> void:
    _camp_stop_medical()
    if is_instance_valid(model_root) and _role == "camp_medico":
        if MedicalSculpt.inspect_interior(model_root, enabled):
            _update_details()

func _camp_pose(progress: float) -> void:
    if model_root == null or not _role.begins_with("camp_"):
        return
    _camp_stop_medical()
    _camp_stop_defense()
    _camp_stop_recon()
    _camp_stop_fx()
    _camp_stop_service()
    if not CampArt.pose_deploy(model_root, progress):
        push_error("CAMP STUDIO: physical deployment pose failed")
        return
    _update_details()

func _camp_stop_service() -> void:
    if _camp_service_tween != null and _camp_service_tween.is_valid():
        _camp_service_tween.kill()
    _camp_service_tween = null
    if model_root != null and _role == "camp_suministros":
        SupplyV43.service_pose(model_root, 0.0)

func _camp_interior(enabled: bool) -> void:
    if model_root == null or _role != "camp_suministros":
        return
    _camp_stop_fx()
    _camp_stop_service()
    if not CampArt.pose_deploy(model_root, 1.0):
        return
    if SupplyV43.inspect_interior(model_root, enabled):
        _update_camera()
        _update_details()

func _camp_service_frame(progress: float) -> void:
    if model_root != null and _role == "camp_suministros":
        SupplyV43.service_pose(model_root, progress)

func _camp_animate_service() -> void:
    if model_root == null or _role != "camp_suministros":
        return
    _camp_stop_fx()
    if not SupplyV43.service_pose(model_root, 0.0):
        return
    _camp_stop_service()
    _camp_service_tween = create_tween()
    _camp_service_tween.tween_method(_camp_service_frame, 0.0, 1.0, 1.6)

func _camp_has_damage_fx() -> bool:
    if not is_instance_valid(model_root):
        return false
    return (_role == "camp_suministros" and model_root.get_meta("suministros_art_v4", 0) == 4) or (_role == "camp_reconocimiento" and model_root.get_meta("recon_art_v3", 0) == 3) or (_role == "camp_defensivo" and model_root.get_meta("defense_art_v3", 0) == 3) or (_role == "camp_medico" and model_root.get_meta("medical_art_v3", 0) == 3)

func _camp_stop_fx() -> void:
    _camp_fx_generation += 1
    if _camp_fx_tween != null and _camp_fx_tween.is_valid():
        _camp_fx_tween.kill()
    _camp_fx_tween = null
    if _camp_has_damage_fx():
        _camp_fx_frame("idle", 0.34)

func _camp_fx_frame(kind: String, progress: float) -> void:
    if model_root != null and _role == "camp_suministros":
        SupplyV43.damage_fx_pose(model_root, kind, progress)
    elif model_root != null and _role == "camp_reconocimiento":
        ReconStation.damage_fx_pose(model_root, kind, progress)
    elif model_root != null and _role == "camp_defensivo":
        DefenseMotion.damage_fx_pose(model_root, kind, progress)
    elif model_root != null and _role == "camp_medico":
        MedicalSculpt.damage_fx_pose(model_root, kind, progress)

func _camp_play_fx(kind: String) -> void:
    if not _camp_has_damage_fx():
        return
    if kind in ["bullet", "blast"] and int(model_root.get_meta("visual_damage", 0)) == 3:
        return
    _camp_stop_medical()
    _camp_stop_defense()
    _camp_stop_recon()
    _camp_stop_service()
    _camp_stop_fx()
    _camp_fx_tween = create_tween()
    var duration: float = 2.8 if kind == "idle" else (1.25 if kind == "blast" else 0.65)
    if kind == "idle":
        _camp_fx_tween.set_loops()
    _camp_fx_tween.tween_method(func(value: float): _camp_fx_frame(kind, value), 0.0, 1.0, duration)

func _camp_destroy_sequence() -> void:
    if not _camp_has_damage_fx():
        return
    if int(model_root.get_meta("visual_damage", 0)) >= 3:
        return
    _camp_stop_medical()
    _camp_stop_defense()
    _camp_stop_recon()
    _camp_stop_service()
    _camp_stop_fx()
    var model_id: int = model_root.get_instance_id()
    var generation: int = _camp_fx_generation
    var kind: String = "bullet" if int(model_root.get_meta("visual_damage", 0)) == 0 else "blast"
    _camp_fx_tween = create_tween()
    _camp_fx_tween.tween_method(func(value: float): _camp_fx_frame(kind, value), 0.0, 1.0, 1.15)
    _camp_fx_tween.tween_callback(func(): _camp_advance_damage.call_deferred(model_id, generation))

func _camp_advance_damage(expected_model: int, generation: int) -> void:
    if not _camp_has_damage_fx() or model_root.get_instance_id() != expected_model or generation != _camp_fx_generation:
        return
    var next: int = mini(3, int(model_root.get_meta("visual_damage", 0)) + 1)
    camp_damage_picker.select(next)
    _camp_display()
    if next < 3:
        _camp_destroy_sequence()
    else:
        _camp_play_fx("idle")

func _update_camera() -> void:
    if _role == "destroyer" and camera != null and destroyer_controls != null:
        camera.projection = Camera3D.PROJECTION_ORTHOGONAL
        camera.size = 3.7 / clampf(_zoom,0.65,2.8)
        camera.position = Vector3(5.3,-2.5,6.3) if _destroyer_underwater else Vector3(5.3,4.2,6.3)
        camera.look_at(Vector3(0,0.40,0),Vector3.UP)
        return
    if _role.begins_with("camp_") and model_root != null and camera != null:
        var focus_y: float = 1.55 if camp_kind == "reconocimiento" else 0.95
        var base_size: float = 11.9 if camp_kind == "reconocimiento" else 9.8
        if camp_kind == "suministros" and model_root.get_meta("suministros_v43_interior_open", false):
            focus_y = 1.70
            base_size = 11.0
        camera.projection = Camera3D.PROJECTION_ORTHOGONAL
        camera.size = base_size / clampf(_zoom, 0.65, 2.8)
        camera.position = Vector3(9.6, 8.1, -10.2)
        camera.look_at(Vector3(0, focus_y, 0), Vector3.UP)
        return
    super._update_camera()

func _reset_camera() -> void:
    if _role.begins_with("camp_"):
        _yaw = -32.0
        _zoom = 1.0
        if model_root != null:
            model_root.rotation_degrees.y = _yaw
        _update_camera()
        return
    super._reset_camera()

func _update_details() -> void:
    if _role.begins_with("camp_") and model_root != null:
        var faction: int = int(model_root.get_meta("visual_faction", -1))
        var lod: int = int(model_root.get_meta("visual_lod", 13))
        var damage: int = int(model_root.get_meta("visual_damage", 0))
        var builder: String = str(CampArt.BUILDERS.get(camp_kind, ""))
        var faction_name: String = "Neutral" if faction < 0 else (
            str(CampStyle.FACTIONS[faction]["name"]))
        var progress: float = float(model_root.get_meta(
            "visual_deploy_progress", 1.0))
        if camp_status != null:
            camp_status.text = ("MODELO 3D REAL · %s · %s · LOD%d · D%d · %s"
                % [camp_kind.capitalize(), faction_name, lod, damage,
                    "Desplegado" if progress >= 0.99 else ("Recogido" if progress <= 0.01 else "En transición")])
        if detail_label != null:
            detail_label.text = "Campamento temporal · constructor " + builder
        if status_label != null:
            status_label.text = ("Sólidos Godot opacos · Sin nuevas reglas "
                + "de curación, defensa, visión o abastecimiento")
        return
    super._update_details()

func _destroyer_studio_select_from_url() -> void:
    if not OS.has_feature("web"):
        return
    var key: String = str(JavaScriptBridge.eval(
        "(new URLSearchParams(window.location.search)).get('unidad') || ''",true))
    if key == "destroyer" and destroyer_controls != null:
        category_picker.select(0)
        _switch_category(0)
        role_picker.select(NavalArt.ROLES.find("destroyer"))
        _display_role("destroyer")
        _studio_select_tab(0)
        # Publish evidence from the instantiated native model, not the URL alone.
        if is_instance_valid(model_root) and _role == "destroyer":
            var info: Dictionary = {
                "role": _role,
                "revision": str(model_root.get_meta("destroyer_art_revision", "")),
                "stage": int(model_root.get_meta("destroyer_damage_stage", -1)),
                "lod": 8 if destroyer_lod_picker.selected == 0 else 13,
                "factions": destroyer_faction_picker.item_count,
                "damage_states": destroyer_damage_picker.item_count,
                "controls_visible": destroyer_controls.is_visible_in_tree(),
                "naval_controls_in_quick_panel": destroyer_controls.get_parent() == studio_quick_panel,
                "generic_animation_hidden": not studio_animation_panel.visible,
                "generic_faction_hidden": not faction_picker.visible,
                "water_visible": is_instance_valid(_destroyer_water) and _destroyer_water.visible
            }
            JavaScriptBridge.eval("window.__tarvokaDestroyerEntry = " + JSON.stringify(info), true)

func _destroyer_restore_generic_faction() -> void:
    if _destroyer_generic_faction_hidden and faction_picker != null:
        faction_picker.visible = _destroyer_generic_faction_previous
    _destroyer_generic_faction_hidden = false

func _refresh_unified_studio_navigation() -> void:
    super._refresh_unified_studio_navigation()
    if destroyer_controls != null:
        _destroyer_studio_sync_visibility()

func _destroyer_studio_sync_visibility() -> void:
    var active: bool = _role == "destroyer" and category_picker != null and category_picker.selected == 0
    if destroyer_controls != null:
        destroyer_controls.visible = active and studio_active_tab == 0
        if active:
            if studio_animation_panel != null:
                studio_animation_panel.hide()
            if studio_overview_label != null:
                studio_overview_label.text = "Destructor naval · V1.23"
            if faction_picker != null:
                if not _destroyer_generic_faction_hidden:
                    _destroyer_generic_faction_previous = faction_picker.visible
                    _destroyer_generic_faction_hidden = true
                faction_picker.hide()
        else:
            _destroyer_restore_generic_faction()
    if is_instance_valid(_destroyer_water):
        _destroyer_water.visible = _role == "destroyer" and category_picker != null and category_picker.selected == 0 and not _destroyer_underwater
    if is_instance_valid(_destroyer_inspection_light):
        _destroyer_inspection_light.visible = active and _destroyer_light_enabled
    if is_instance_valid(_destroyer_lower_fill):
        _destroyer_lower_fill.visible = active and _destroyer_underwater

func _display_role(role: String) -> void:
    _destroyer_studio_stop()
    if role != "destroyer":
        _destroyer_restore_generic_faction()
    if role != "destroyer" or _role != "destroyer":
        _destroyer_studio_restore_floor()
        _destroyer_underwater = false
        _destroyer_light_enabled = false
    super._display_role(role)
    if role == "destroyer" and destroyer_controls != null:
        _destroyer_studio_rebuild()
    _destroyer_studio_sync_visibility()

func _destroyer_studio_rebuild() -> void:
    if _role != "destroyer" or destroyer_controls == null or preview == null:
        return
    _destroyer_studio_stop()
    if is_instance_valid(model_root):
        if model_root.get_parent() != null:
            model_root.get_parent().remove_child(model_root)
        model_root.queue_free()
    var lod: int = 8 if destroyer_lod_picker.selected == 0 else 13
    model_root = NavalArt.build("destroyer",destroyer_faction_picker.selected-1,true,lod)
    if model_root == null:
        return
    preview.add_child(model_root)
    model_root.rotation_degrees.y = _yaw
    _destroyer_studio_damage_changed(destroyer_damage_picker.selected)
    _update_camera()
    _update_details()

func _destroyer_studio_option_changed(_index: int) -> void:
    _destroyer_studio_rebuild()

func _destroyer_studio_damage_changed(stage: int) -> void:
    if _role != "destroyer" or not is_instance_valid(model_root):
        return
    _destroyer_studio_stop()
    destroyer_damage_picker.select(clampi(stage,0,4))
    _destroyer_fx_time = 0.0
    NavalArt.set_destroyer_damage(model_root,stage,0.45 if stage == 4 else 0.0)
    if destroyer_status != null:
        destroyer_status.text = "Modelo nativo V1.23 · D%d" % stage

func _destroyer_studio_stop() -> void:
    if _destroyer_studio_tween != null and _destroyer_studio_tween.is_running():
        _destroyer_studio_tween.kill()
    if _role == "destroyer" and is_instance_valid(model_root):
        NavalArt.restore_model_pose("destroyer",model_root)

func _destroyer_studio_frame(kind: String, phase: float) -> void:
    if _role != "destroyer" or not is_instance_valid(model_root):
        return
    if kind == "sink":
        NavalArt.set_destroyer_damage(model_root,4,phase)
    else:
        NavalArt.pose_model("destroyer",model_root,kind,phase)

func _destroyer_studio_restore_floor() -> void:
    for floor_mesh in _destroyer_hidden_floor:
        if is_instance_valid(floor_mesh):
            floor_mesh.show()
    _destroyer_hidden_floor.clear()

func _destroyer_studio_view(underwater: bool) -> void:
    if _role != "destroyer":
        return
    _destroyer_studio_restore_floor()
    _destroyer_underwater = underwater
    if underwater and is_instance_valid(preview) and is_instance_valid(model_root):
        for candidate in preview.find_children("*","MeshInstance3D",true,false):
            var floor_mesh: MeshInstance3D = candidate as MeshInstance3D
            if floor_mesh == null or floor_mesh == _destroyer_water or model_root.is_ancestor_of(floor_mesh):
                continue
            var label: String = str(floor_mesh.name).to_lower()
            if floor_mesh.visible and (floor_mesh.mesh is PlaneMesh or label.contains("suelo") or label.contains("floor") or label.contains("ground")):
                _destroyer_hidden_floor.append(floor_mesh)
                floor_mesh.hide()
    _destroyer_studio_sync_visibility()
    _update_camera()

func _destroyer_studio_light(enabled: bool) -> void:
    if _role != "destroyer":
        return
    _destroyer_light_enabled = enabled
    _destroyer_studio_sync_visibility()

func _destroyer_studio_play(kind: String) -> void:
    if _role != "destroyer" or not is_instance_valid(model_root):
        return
    _destroyer_studio_stop()
    if kind == "sink":
        _destroyer_studio_view(false)
        destroyer_damage_picker.select(4)
        _destroyer_studio_frame(kind,0.0)
    elif int(model_root.get_meta("destroyer_damage_stage",0)) >= 3:
        destroyer_status.text = "Reparar el modelo antes de iniciar otra animación."
        return
    _destroyer_studio_tween = create_tween()
    _destroyer_studio_tween.tween_method(func(v: float): _destroyer_studio_frame(kind,v),0.0,1.0,
        2.7 if kind == "sink" else 1.5)
    _destroyer_studio_tween.finished.connect(func():
        if _role == "destroyer" and is_instance_valid(model_root):
            if kind != "sink":
                NavalArt.restore_model_pose("destroyer",model_root)
            else:
                destroyer_status.text = "Modelo nativo V1.23 · D4 · Hundido")

func _process(delta: float) -> void:
    super._process(delta)
    if _role == "destroyer" and is_instance_valid(model_root):
        _destroyer_fx_time = fposmod(_destroyer_fx_time+delta*0.65,1.0)
        NavalArt.animate_destroyer_damage(model_root,_destroyer_fx_time)
